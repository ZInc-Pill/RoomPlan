                </div>
                <div>
                  <div className="text-xs font-bold text-white leading-tight">
                    {selectedItem 
                      ? (selectedItemType?.name || 'Selected Object') 
                      : selectedWall 
                        ? 'Wall'

                  </div>

                      : 'text-slate-400 hover:text-white'

                  <Magnet className="w-3.5 h-3.5 shrink-0" />

                    !snapMode
                      ? 'bg-indigo-600 text-white shadow-sm'
                      : 'text-slate-400 hover:text-white'
                  }`}
                  title="Free Move (Secondary) - continuous smooth movement"
                >
                  <Move className="w-3.5 h-3.5 shrink-0" />
                  <span>Free Move</span>
                </button>
              </div>

                    className="flex items-center gap-1 px-2 py-1 rounded-lg text-xs font-semibold text-slate-300 hover:text-white hover:bg-slate-700 active:scale-95 transition-all"

                    <RotateCcw className="w-3.5 h-3.5 text-indigo-400" />

                  <div className="w-px h-3.5 bg-slate-700 mx-0.5" />

                    className="flex items-center gap-1 px-2 py-1 rounded-lg text-xs font-semibold text-slate-300 hover:text-white hover:bg-slate-700 active:scale-95 transition-all"
                    title="Rotate +45° (Keyboard: R)"
                  >
                    <RotateCw className="w-3.5 h-3.5 text-indigo-400" />
                    <span>+45°</span>
                  </button>
                </div>
              )}

              {/* Quick Action: Delete */}

                onClick={selectedItem ? onDeleteItem : selectedWall ? onDeleteWall : onDeleteFloor}
                className="p-2 rounded-xl bg-rose-500/20 hover:bg-rose-500/30 text-rose-300 border border-rose-500/30 active:scale-95 transition-all"
                title="Delete Selected (Del)"
              >
